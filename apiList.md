# DEVTinder APIs

# authRouter
 - POST /signup
 - POST /login
 - POST /logout

# profileRouter
 - GET /profile/view
 - PATCH /profile/edit
 - PATCH /profile/password

# connectionRequestRouter
 - POST /request/send/:status/:userId
 - POST /request/review/:status/:requestId

# userRouter
 - GET /requests/received
 - GET /connections
 - GET /feed - gets profiles of other users on platform
